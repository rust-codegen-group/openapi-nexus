package runtime;

import com.google.gson.Gson;
import com.google.gson.JsonElement;
import com.google.gson.JsonObject;
import com.google.gson.JsonParser;
import com.google.gson.JsonSyntaxException;
import com.google.gson.TypeAdapter;
import com.google.gson.TypeAdapterFactory;
import com.google.gson.annotations.SerializedName;
import com.google.gson.internal.LazilyParsedNumber;
import com.google.gson.internal.bind.JsonTreeReader;
import com.google.gson.internal.bind.JsonTreeWriter;
import com.google.gson.reflect.TypeToken;
import com.google.gson.stream.JsonReader;
import com.google.gson.stream.JsonToken;
import com.google.gson.stream.JsonWriter;
import java.io.IOException;
import java.io.StringReader;
import java.io.StringWriter;
import java.lang.reflect.Field;
import java.lang.reflect.Modifier;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

/** Internal flat composition; generated subclasses keep member codecs statically typed. */
public abstract class AllOfAdapterFactory<M> implements TypeAdapterFactory {
    private final Class<M> modelClass;
    private final Member[] members;

    protected static final class Member {
        private final String[] declared;
        private final Class<?>[] sources;
        private Member(String[] declared, Class<?>[] sources) {
            this.declared = declared;
            this.sources = sources;
        }
    }

    protected static Member member(String[] declared, Class<?>... sources) {
        return new Member(declared, sources);
    }

    protected AllOfAdapterFactory(Class<M> modelClass, Member... members) {
        this.modelClass = modelClass;
        this.members = members;
    }

    protected abstract M decode(Gson gson, List<JsonReader> readers) throws IOException;
    protected abstract List<JsonElement> encode(Gson gson, M value, JsonWriter context) throws IOException;

    protected static <V> JsonElement tree(Gson gson, V value, TypeToken<V> type, JsonWriter context) throws IOException {
        JsonTreeWriter writer = new JsonTreeWriter();
        writer.setSerializeNulls(context.getSerializeNulls());
        writer.setHtmlSafe(context.isHtmlSafe());
        writer.setStrictness(context.getStrictness());
        gson.getAdapter(type).write(writer, value);
        return writer.get();
    }

    @Override
    @SuppressWarnings("unchecked")
    public final <T> TypeAdapter<T> create(Gson gson, TypeToken<T> type) {
        if (type.getRawType() != modelClass) { return null; }
        List<Set<String>> names = new ArrayList<>();
        Set<String> allNames = new HashSet<>();
        for (Member member : members) {
            Set<String> known = new HashSet<>(Arrays.asList(member.declared));
            for (Class<?> source : member.sources) {
                for (Field field : source.getDeclaredFields()) {
                    if (Modifier.isStatic(field.getModifiers()) || Modifier.isTransient(field.getModifiers())) { continue; }
                    SerializedName annotation = field.getAnnotation(SerializedName.class);
                    if (annotation == null) { known.add(gson.fieldNamingStrategy().translateName(field)); }
                    else {
                        known.add(annotation.value());
                        known.addAll(Arrays.asList(annotation.alternate()));
                    }
                }
            }
            names.add(known);
            allNames.addAll(known);
        }
        TypeAdapter<JsonElement> trees = gson.getAdapter(JsonElement.class);
        TypeAdapter<M> adapter = new TypeAdapter<>() {
            @Override public M read(JsonReader reader) throws IOException {
                if (members.length == 0) { return decode(gson, List.of(reader)); }
                if (reader.peek() == JsonToken.NULL) { reader.nextNull(); return null; }
                if (reader.peek() != JsonToken.BEGIN_OBJECT) { throw new JsonSyntaxException("Expected " + modelClass.getSimpleName() + " object"); }
                StringWriter[] buffers = new StringWriter[members.length];
                JsonWriter[] projections = new JsonWriter[members.length];
                for (int i = 0; i < members.length; i++) {
                    buffers[i] = new StringWriter();
                    projections[i] = new JsonWriter(buffers[i]);
                    projections[i].setSerializeNulls(true);
                    projections[i].setStrictness(reader.getStrictness());
                    projections[i].beginObject();
                }
                reader.beginObject();
                while (reader.hasNext()) {
                    String key = reader.nextName();
                    List<JsonWriter> targets = new ArrayList<>();
                    for (int i = 0; i < members.length; i++) {
                        if (!allNames.contains(key) || names.get(i).contains(key)) {
                            projections[i].name(key);
                            targets.add(projections[i]);
                        }
                    }
                    // Copy tokens, not a JsonObject: ordering, duplicate keys and
                    // numeric lexemes must reach the configured member delegates.
                    copy(reader, targets);
                }
                reader.endObject();
                JsonReader[] replays = new JsonReader[members.length];
                for (int i = 0; i < members.length; i++) {
                    projections[i].endObject();
                    if (reader instanceof JsonTreeReader) { replays[i] = new JsonTreeReader(JsonParser.parseString(buffers[i].toString())); }
                    else { replays[i] = new JsonReader(new StringReader(buffers[i].toString())); }
                    replays[i].setStrictness(reader.getStrictness());
                }
                return decode(gson, Arrays.asList(replays));
            }

            @Override public void write(JsonWriter writer, M value) throws IOException {
                if (value == null) { writer.nullValue(); return; }
                List<JsonElement> values = encode(gson, value, writer);
                JsonElement output;
                if (members.length == 0) { output = values.get(0); }
                else {
                    JsonObject merged = new JsonObject();
                    for (JsonElement member : values) {
                        if (member == null || member.isJsonNull()) { continue; }
                        if (!member.isJsonObject()) { throw new JsonSyntaxException("Expected allOf member object"); }
                        for (Map.Entry<String, JsonElement> entry : member.getAsJsonObject().entrySet()) {
                            JsonElement previous = merged.get(entry.getKey());
                            if (previous != null && !previous.toString().equals(entry.getValue().toString())) {
                                throw new IllegalArgumentException("Conflicting allOf property: " + entry.getKey());
                            }
                            merged.add(entry.getKey(), entry.getValue());
                        }
                    }
                    output = merged;
                }
                // Named member codecs already applied their own null policy.
                // Retain explicit map/raw nulls without leaking writer state.
                boolean serializeNulls = writer.getSerializeNulls();
                writer.setSerializeNulls(true);
                try { trees.write(writer, output); }
                finally { writer.setSerializeNulls(serializeNulls); }
            }
        };
        // The raw-type identity above proves this cast; allOf models are not generic.
        return (TypeAdapter<T>) adapter;
    }

    private static void copy(JsonReader reader, List<JsonWriter> targets) throws IOException {
        switch (reader.peek()) {
            case BEGIN_OBJECT:
                reader.beginObject();
                for (JsonWriter target : targets) { target.beginObject(); }
                while (reader.hasNext()) {
                    String key = reader.nextName();
                    for (JsonWriter target : targets) { target.name(key); }
                    copy(reader, targets);
                }
                reader.endObject();
                for (JsonWriter target : targets) { target.endObject(); }
                break;
            case BEGIN_ARRAY:
                reader.beginArray();
                for (JsonWriter target : targets) { target.beginArray(); }
                while (reader.hasNext()) { copy(reader, targets); }
                reader.endArray();
                for (JsonWriter target : targets) { target.endArray(); }
                break;
            case NUMBER:
                Number number = new LazilyParsedNumber(reader.nextString());
                for (JsonWriter target : targets) { target.value(number); }
                break;
            case STRING:
                String string = reader.nextString();
                for (JsonWriter target : targets) { target.value(string); }
                break;
            case BOOLEAN:
                boolean bool = reader.nextBoolean();
                for (JsonWriter target : targets) { target.value(bool); }
                break;
            case NULL:
                reader.nextNull();
                for (JsonWriter target : targets) { target.nullValue(); }
                break;
            default: throw new JsonSyntaxException("Expected JSON value");
        }
    }
}
